        // Configuration
        const API_BASE_URL = '/api';
        const UPAYU_API_BASE_URL = '[https://api.upayu.co.id/v1](https://api.upayu.co.id/v1)';
        const UPAYU_WEBHOOK_SECRET = 'your_webhook_secret_here'; // Replace with your actual secret
        // DOM Elements
        const paymentForm = document.getElementById('paymentForm');
        const paymentStatus = document.getElementById('paymentStatus');
        const captureBtn = document.getElementById('captureBtn');
        const transactionIdInput = document.getElementById('transactionId');
        const webhookLog = document.getElementById('webhookLog');
        // Current payment state
        let currentPayment = null;
        // Initialize the app
        document.addEventListener('DOMContentLoaded', function() {
            // Set up event listeners
            paymentForm.addEventListener('submit', handleCreatePayment);
            captureBtn.addEventListener('click', handleCapturePayment);
            // Load any existing payment status
            loadPaymentStatus();
        });
        // Create Payment Handler
        async function handleCreatePayment(e) {
            e.preventDefault();
            const amount = document.getElementById('amount').value;
            const currency = document.getElementById('currency').value;
            const paymentMethod = document.getElementById('paymentMethod').value;
            const metadata = document.getElementById('metadata').value;
            try {
                const response = await fetch`${API_BASE_URL}/payments`, {
                    method: 'POST',
                    headers: {
                        'Content-Type': 'application/json',
                    },
                    body: JSON.stringify({
                        amount,
                        currency,
                        paymentMethod,
                        metadata: metadata ? JSON.parse(metadata) : {}
                    })
                });
                const data = await response.json();
                if (response.ok) {
                    currentPayment = data.payment;
                    paymentStatus.innerHTML = `
                        <div class="d-flex justify-content-between">
                            <div>
                                <h6>Payment Created</h6>
                                <p>Amount: ${data.payment.amount} ${data.payment.currency}</p>
                                <p>Transaction ID: ${data.payment.transaction_id}</p>
                                <p>Status: <span class="payment-status status-pending">Pending</span></p>
                            </div>
                            <a href="${data.redirectUrl}" target="_blank" class="btn btn-primary">Pay Now</a>
                        </div>
                    `;
                    // Enable capture button
                    captureBtn.disabled = false;
                    transactionIdInput.value = data.payment.transaction_id;
                    // Add to webhook log
                    addWebhookLogEntry('Payment created', data.payment);
                } else {
                    throw new Error(data.error || 'Failed to create payment');
                }
            } catch (error) {
                paymentStatus.innerHTML = `
                    <div class="alert alert-danger">
                        Error: ${error.message}
                    </div>
                `;
                console.error('Payment creation error:', error);
            }
        }
        // Capture Payment Handler
        async function handleCapturePayment() {
            const transactionId = transactionIdInput.value;
            if (!transactionId) {
                alert('Please enter a transaction ID');
                return;
            }
            try {
                const response = await fetch`${API_BASE_URL}/payments/${transactionId}/capture`, {
                    method: 'POST'
                });
                const data = await response.json();
                if (response.ok) {
                    paymentStatus.innerHTML = `
                        <div class="d-flex justify-content-between">
                            <div>
                                <h6>Payment Captured</h6>
                                <p>Amount: ${data.payment.amount} ${data.payment.currency}</p>
                                <p>Transaction ID: ${data.payment.transaction_id}</p>
                                <p>Status: <span class="payment-status status-completed">Completed</span></p>
                            </div>
                        </div>
                    `;
                    // Add to webhook log
                    addWebhookLogEntry('Payment captured', data.payment);
                    // Update current payment status
                    currentPayment.status = 'completed';
                } else {
                    throw new Error(data.error || 'Failed to capture payment');
                }
            } catch (error) {
                paymentStatus.innerHTML = `
                    <div class="alert alert-danger">
                        Error: ${error.message}
                    </div>
                `;
                console.error('Capture error:', error);
            }
        }
