document.addEventListener('DOMContentLoaded', function() {
    // DOM elements
    const gridContainer = document.getElementById('grid-container');
    const thresholdValueDisplay = document.getElementById('threshold-value');
    const thresholdButtons = {
        'threshold-10': document.getElementById('threshold-10'),
        'threshold-100': document.getElementById('threshold-100'),
        'threshold-1000': document.getElementById('threshold-1000')
    };
    // State
    let currentThreshold = 10;
    let draggedItem = null;
    let offsetX, offsetY;
    // Initialize the grid with 10 items
    updateGrid(10);
    // Set up event listeners for threshold buttons
    for (const [id, button] of Object.entries(thresholdButtons)) {
        button.addEventListener('click', function() {
            const threshold = parseInt(id.replace('threshold-', ''));
            currentThreshold = threshold;
            thresholdValueDisplay.textContent = threshold;
            updateGrid(threshold);
        });
    }
    // Update grid based on current threshold
    function updateGrid(threshold) {
        // Clear existing grid items
        gridContainer.innerHTML = '';
        // Create new grid items based on threshold
        const items = [];
        for (let i = 0; i < threshold; i++) {
            const item = document.createElement('div');
            item.className = 'grid-item';
            item.textContent = i + 1;
            [item.dataset.id](http://item.dataset.id) = i;
            item.addEventListener('mousedown', startDrag);
            items.push(item);
        }
        // Add items to grid
        gridContainer.append(...items);
    }
    // Drag and drop functionality
    function startDrag(e) {
        if ([e.target](http://e.target).classList.contains('grid-item')) {
            draggedItem = [e.target](http://e.target);
            draggedItem.classList.add('dragging');
            // Calculate offset from mouse to element
            const rect = draggedItem.getBoundingClientRect();
            offsetX = e.clientX - rect.left;
            offsetY = e.clientY - [rect.top](http://rect.top);
            // Add event listeners for dragging
            document.addEventListener('mousemove', drag);
            document.addEventListener('mouseup', stopDrag);
            document.addEventListener('mouseleave', stopDrag);
            e.preventDefault();
        }
    }
    function drag(e) {
        if (!draggedItem) return;
        // Calculate new position
        let x = e.clientX - offsetX;
        let y = e.clientY - offsetY;
        // Constrain to grid container
        const containerRect = gridContainer.getBoundingClientRect();
        x = Math.max(10, Math.min(x, containerRect.width - 50));
        y = Math.max(10, Math.min(y, containerRect.height - 50));
        // Update position
        [draggedItem.style](http://draggedItem.style).left = (x - 25) + 'px';
        [draggedItem.style.top](http://draggedItem.style.top) = (y - 25) + 'px';
    }
    function stopDrag() {
        if (!draggedItem) return;
        // Snap to grid position
        const rect = gridContainer.getBoundingClientRect();
        const gridWidth = rect.width / Math.ceil(Math.sqrt(currentThreshold));
        const gridHeight = rect.height / Math.ceil(Math.sqrt(currentThreshold));
        const x = Math.round((draggedItem.offsetLeft + 25) / gridWidth) * gridWidth - 25;
        const y = Math.round((draggedItem.offsetTop + 25) / gridHeight) * gridHeight - 25;
        [draggedItem.style](http://draggedItem.style).left = x + 'px';
        [draggedItem.style.top](http://draggedItem.style.top) = y + 'px';
        // Remove dragging class
        draggedItem.classList.remove('dragging');
        draggedItem = null;
        // Remove event listeners
        document.removeEventListener('mousemove', drag);
        document.removeEventListener('mouseup', stopDrag);
        document.removeEventListener('mouseleave', stopDrag);
    }
    // Add CSS for absolute positioning
    const style = document.createElement('style');
    style.textContent = `
        .grid-item {
            position: absolute;
            width: 50px;
            height: 50px;
        }
    `;
    document.head.appendChild(style);
});
