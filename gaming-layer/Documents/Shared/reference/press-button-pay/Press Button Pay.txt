    function updateGrid() {
      // Clear existing spheres
      spheres.forEach(sphere => scene.remove(sphere));
      spheres = [];
      // Calculate grid dimensions
      let rows, cols;
      const count = Math.min(currentSphereCount, 256);
      switch (currentGridLayout) {
        case 'pi':
          rows = Math.floor(Math.sqrt(count / Math.PI));
          cols = Math.ceil(rows * Math.PI);
          break;
        case 'golden':
          const phi = (1 + Math.sqrt(5)) / 2;
          rows = Math.floor(Math.sqrt(count / phi));
          cols = Math.ceil(rows * phi);
          break;
        default: // square
          rows = Math.floor(Math.sqrt(count));
          cols = Math.ceil(count / rows);
      }
      // Create spheres
      for (let i = 0; i < count; i++) {
        let row = Math.floor(i / cols);
        let col = i % cols;
        let x, y, z;
        switch (currentGridLayout) {
          case 'pi':
            const angle = (i / count) _Math.PI_ 2;
            x = Math.cos(angle) _(cols_ currentSpacing / 2);
            y = Math.sin(angle) _(rows_ currentSpacing / 2);
            z = 0;
            break;
          case 'golden':
            const goldenAngle = (i / count) _Math.PI_ 2 * 0.618;
            const radius = Math.sqrt(i) * currentSpacing / 2;
            x = Math.cos(goldenAngle) * radius;
            y = Math.sin(goldenAngle) * radius;
            z = 0;
            break;
          default: // square
            x = (col - cols / 2) * currentSpacing;
            y = (row - rows / 2) * currentSpacing;
            z = 0;
        }
        // Create sphere
        const geometry = new THREE.SphereGeometry(currentSphereSize, 32, 32);
        const material = new THREE.MeshStandardMaterial({
          color: getRandomColor(),
          roughness: 0.3,
          metalness: 0.5
        });
        const sphere = new THREE.Mesh(geometry, material);
        sphere.position.set(x, y, z);
        sphere.userData = {
          originalPosition: sphere.position.clone(),
          originalSize: currentSphereSize,
          id: i,
          layout: currentGridLayout,
          value: Math.floor(Math.random() * 10) + 1 // Random "value" for "earning"
        };
        scene.add(sphere);
        spheres.push(sphere);
      }
    }
