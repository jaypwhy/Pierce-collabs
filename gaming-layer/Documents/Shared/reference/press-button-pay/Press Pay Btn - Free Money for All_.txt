  <script>
    // Game state
    let money = 0;
    let clicks = 0;
    let unlocks = 0;
    let upgrades = 0;
    let clickMultiplier = 1;
    let isPaying = false;
    // DOM elements
    const moneyDisplay = document.getElementById('money-display');
    const clickBtn = document.getElementById('click-btn');
    const payBtn = document.getElementById('pay-btn');
    const unlockBtn = document.getElementById('unlock-btn');
    const upgradeBtn = document.getElementById('upgrade-btn');
    const rewardDiv = document.getElementById('reward');
    const statsDiv = document.getElementById('stats');
    // Update UI
    function updateUI() {
      moneyDisplay.textContent = `$${money}`;
      statsDiv.textContent = `Clicks: ${clicks} | Unlocks: ${unlocks} | Upgrades: ${upgrades}`;
    }
    // Click button (free)
    clickBtn.addEventListener('click', () => {
      money += clickMultiplier;
      clicks++;
      showReward`+$${clickMultiplier} (Click!)`);
      updateUI();
    });
    // Pay button (costs money but gives more)
    payBtn.addEventListener('click', () => {
      if (money >= 10) {
        money -= 10;
        money += 50;
        clicks++;
        showReward`You paid $10 and got $50! (Net +$40)`);
        updateUI();
      } else {
        showReward("Not enough money! (Go click more buttons.)");
      }
    });
    // Unlock new button
    unlockBtn.addEventListener('click', () => {
      if (money >= 50) {
        money -= 50;
        unlocks++;
        unlockBtn.disabled = true;
        const newBtn = document.createElement('button');
        newBtn.textContent = "Secret Button (Click 100x for $1000)";
        [newBtn.style](http://newBtn.style).backgroundColor = "#9C27B0";
        [newBtn.style](http://newBtn.style).fontSize = "1rem";
        newBtn.addEventListener('click', () => {
          if (clicks >= 100) {
            money += 1000;
            clicks = 0;
            showReward`+$1000! (But it's just a JPG of a banana.)`);
            updateUI();
          } else {
            showReward`Click 100 times first! (You've clicked ${clicks}/100)`);
          }
        });
        document.querySelector('.button-container').appendChild(newBtn);
        updateUI();
      } else {
        showReward("Not enough money! (Click more buttons.)");
      }
    });
    // Upgrade (double clicks)
    upgradeBtn.addEventListener('click', () => {
      if (money >= 100) {
        money -= 100;
        upgrades++;
        clickMultiplier *= 2;
        upgradeBtn.disabled = true;
        showReward`Upgrade activated! Clicks now give +$${clickMultiplier}!`);
        updateUI();
      } else {
        showReward("Not enough money! (Click more buttons.)");
      }
    });
    // Show reward (with humor)
    function showReward(message) {
      rewardDiv.textContent = message;
      [rewardDiv.style](http://rewardDiv.style).display = 'block';
      // Random funny reward (e.g., banana, meme, etc.)
      const funnyRewards = [
        "🍌 (A banana. It's just a banana.)",
        "🎉 (Congrats! You got a virtual high-five.)",
        "💰 (You found a $10 bill in a sock. Again.)",
        "🤡 (You got a clown nose. Wear it proudly.)"
      ];
      rewardDiv.innerHTML = `<strong>${message}</strong><br>${funnyRewards[Math.floor(Math.random() * funnyRewards.length)]}`;
      setTimeout(() => {
        [rewardDiv.style](http://rewardDiv.style).display = 'none';
      }, 3000);
    }
    // Initialize
    updateUI();
