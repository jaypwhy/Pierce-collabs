name = db.Column(db.String(80), nullable=False)

#### **B. Update the** `add_points` **Route**

python
  
⌄
⌄
from models import db, User
@app.route('/add-points', methods=['POST'])
def add_points():
amount = request.form.get('amount')
user_id = 1 # Replace with dynamic user selection (e.g., from session)
# Fetch the user
user = User.query.get(user_id)
if not user:
return "User not found.", 404
# Update balance (add the amount to the existing balance)
new_balance = str(int(user.balance) + int(amount))
user.balance = new_balance
db.session.commit()
print(f"Updated user {user_id}'s balance to {new_balance}.")
return redirect(url_for('home'))

import pytest
from app import app
from models import db, User
@pytest.fixture
def client():
app.config['TESTING'] = True
app.config['SQLALCHEMY_DATABASE_URI'] = 'sqlite:///:memory:'
with app.test_client() as client:
with app.app_context():
db.create_all()
# Add a test user

db.session.add(test_user)
db.session.commit()
yield client
db.session.remove()
